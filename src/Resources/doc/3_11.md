### 3.11 onValidate callback

Use `onValidate` to run extra logic after validation:

```js
$('form#user').jsFormValidator({
    onValidate: function(errors, event) {
        // event is the form submission event
        // errors is an object that looks like:
        /*
        errors = {
            element_1_id: {
                source_1_id: ['error_1', 'error_2', 'error_3'],
                source_2_id: ['error_4', 'error_5']
            },
            element_2_id: {
                ...
            }
        }
        */

        // Your code here.
    }
});
```

Pure JavaScript:

```js
var field = document.getElementById('user');
FpJsFormValidator.customize(field, {
    onValidate: function(errors, event) {
        // Your code here.
    }
});
```

`onValidate` should be defined on the parent form element.
